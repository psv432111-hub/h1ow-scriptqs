local MOD_GROUP_IDS = { 925309458, 33991282, 7431102 }

local function hasModSymbols(name) for _, sym in ipairs(MOD_SYMBOLS) do if string.find(name, sym, 1, true) then return true end end return false end

local function checkAdmin(player)
    local nameLower = player.Name:lower()
    local displayLower = player.DisplayName:lower()
    if MOD_USERNAMES[nameLower] or MOD_USERNAMES[displayLower] then return true, "Known Mod" end
    if hasModSymbols(player.Name) or hasModSymbols(player.DisplayName) then return true, "Mod Symbol" end
    for _, groupId in ipairs(MOD_GROUP_IDS) do
        local ok, inGroup = pcall(function() return player:IsInGroup(groupId) end)
        if ok and inGroup then
            local rankOk, rank = pcall(function() return player:GetRankInGroup(groupId) end)
            local roleOk, role = pcall(function() return player:GetRoleInGroup(groupId) end)
            local roleName = (roleOk and role) or "GroupMember"
            if rankOk and rank >= MOD_MIN_RANK then return true, roleName .. " [" .. tostring(rank) .. "]" end
        end
    end
    return false, nil
end

local adminCache = {} local CACHE_TTL = 30
local function getCachedCheck(player)
    local cached = adminCache[player.UserId]

local function spectateDetect()
    local remotes = ReplicatedStorage:FindFirstChild("AdminRemotes")
    if not remotes then return end
    local player = LocalPlayer
    local spec = remotes:FindFirstChild("AdminSpectateNew")
    if spec and spec:IsA("RemoteEvent") then
        spec.OnClientEvent:Connect(function(...)
            local args = {...} local sn, tn = nil, nil
            for _, v in ipairs(args) do if type(v) == "string" then if v == player.Name then tn = v else sn = v end end end
            if tn == player.Name and sn then currentSpectator = sn pcall(updateStaffList) else if currentSpectator then currentSpectator = nil pcall(updateStaffList) end end
        end)
    end
    task.spawn(function()
        while staffScreenGui and staffScreenGui.Parent do
            task.wait(0.05)
            local myChar = LocalPlayer.Character
            if not myChar then if currentSpectator then currentSpectator = nil pcall(updateStaffList) end continue end
            local subject = Camera.CameraSubject local isMe = false
            if subject then if subject == myChar then isMe = true elseif subject:IsDescendantOf(myChar) then isMe = true end end
            local specActive = remotes:GetAttribute("SpecActive")
            if specActive and isMe then if not currentSpectator then currentSpectator = "Администратор зашел в спек" pcall(updateStaffList) end
            else if currentSpectator then currentSpectator = nil pcall(updateStaffList) end end
        end
    end)
end

local function IsPlayerAlive(player)
    if not player then return false end
    if not player.Parent then return false end
    local char = player.Character
    if not char then return false end
    if not char.Parent then return false end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    if humanoid.Health <= 0 then return false end
    local state = humanoid:GetState()
    if state == Enum.HumanoidStateType.Dead then return false end
    if state == Enum.HumanoidStateType.Physics then return false end
    local knocked = char:FindFirstChild("Knocked") or char:FindFirstChild("Downed") or char:FindFirstChild("IsDown") or char:FindFirstChild("GRABBING_CONSTRAINT")
    if knocked then return false end
    if humanoid.WalkSpeed == 0 and humanoid.Health > 0 then return false end
    if humanoid.PlatformStand then return false end
    return true
end

-- ========== HITBOX ==========
local function applyHitbox(plr)
    if not plr or plr == LocalPlayer then return end
    local char = plr.Character if not char then return end
    local head = char:FindFirstChild("Head")
    if not head then return end
    if not _S.OriginalPartSizes[head] then
        _S.OriginalPartSizes[head] = { Size = head.Size, Transparency = head.Transparency, Color = head.Color, Material = head.Material, CanCollide = head.CanCollide, Massless = head.Massless }
    end
    pcall(function()
        local s = _S.HitboxSize
        head.Size = Vector3.new(s, s, s)
        head.Transparency = 1
        head.CanCollide = false
        head.Massless = true
    end)
end

local function resetHitbox(plr)
    if not plr or plr == LocalPlayer then return end
    local char = plr.Character if not char then return end
    local head = char:FindFirstChild("Head")
    if head and _S.OriginalPartSizes[head] then
        local d = _S.OriginalPartSizes[head]
        pcall(function()
            head.Size = d.Size head.Transparency = d.Transparency head.Color = d.Color
            head.Material = d.Material head.CanCollide = d.CanCollide head.Massless = d.Massless
        end)
        _S.OriginalPartSizes[head] = nil
    end
end

local function GetPlayerFromPart(part)
    if not part then return nil end
    local m = part:FindFirstAncestorWhichIsA("Model") if not m then return nil end
    if m:FindFirstChildWhichIsA("Humanoid") then for _, plr in pairs(Players:GetPlayers()) do if plr.Character == m then return plr end end end
    return nil
end

local function HasLineOfSight(tp)
    if not tp then return false end
    local c = LocalPlayer.Character if not c then return false end
    local o = Camera.CFrame.Position
    local d = (tp.Position - o)
    local p = RaycastParams.new() p.FilterType = Enum.RaycastFilterType.Exclude p.FilterDescendantsInstances = {c} p.IgnoreWater = true
    if _S.CheckFriend then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and IsFriend(plr) and plr.Character then table.insert(p.FilterDescendantsInstances, plr.Character) end
        end
    end
    local r = workspace:Raycast(o, d.Unit * (d.Magnitude + 1), p)
    if not r then return true end
    local hp = GetPlayerFromPart(r.Instance)
    if hp and hp.Character == tp:FindFirstAncestorWhichIsA("Model") then return true end
    return false
end

    if _S.TriggerbotConnection then _S.TriggerbotConnection:Disconnect() _S.TriggerbotConnection = nil end

        _S.TriggerbotConnection = RunService.Heartbeat:Connect(function()
            pcall(function()

                if not currentTool then return end
                local toolName = currentTool.Name
                local isWeaponEnabled = false
                if _S.WeaponsEnabled then
                    if _S.WeaponsEnabled[toolName] then isWeaponEnabled = true
                    else local bracketName = "[" .. toolName .. "]" if _S.WeaponsEnabled[bracketName] then isWeaponEnabled = true end end
                end
                if not isWeaponEnabled then return end
                if _S.TriggerbotRMBOnly and not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
                local mouse = LocalPlayer:GetMouse()
                local target = mouse.Target
                if target then
                    local targetPart = target
                    local targetParent = targetPart.Parent
                    local targetPlayer = nil
                    while targetParent and targetParent ~= game do
                        if targetParent:IsA("Model") then
                            local humanoid = targetParent:FindFirstChildOfClass("Humanoid")
                            if humanoid then for _, plr in pairs(Players:GetPlayers()) do if plr.Character == targetParent then targetPlayer = plr break end end break end
                        end
                        targetParent = targetParent.Parent
                    end
                    if targetPlayer and targetPlayer ~= LocalPlayer then
                        if not IsLegitTarget(targetPlayer) then return end
                        if _S.CheckDead and not IsPlayerAlive(targetPlayer) then return end
                        if _S.CheckWall and not HasLineOfSight(targetPart) then return end
                        task.wait(_S.TriggerbotDelay + math.random() * 0.003)
                        if _S.CheckDead and not IsPlayerAlive(targetPlayer) then return end
                        mouse1press() task.wait(0.025 + math.random() * 0.01) mouse1release()
                    end
                end
            end)
        end)
    end
end

local FlingActive = false
local function SkidFling(TargetPlayer)
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter then return end
    local THumanoid, TRootPart, THead, Accessory, Handle
    if TCharacter:FindFirstChildOfClass("Humanoid") then THumanoid = TCharacter:FindFirstChildOfClass("Humanoid") end
    if THumanoid and THumanoid.RootPart then TRootPart = THumanoid.RootPart end
    if TCharacter:FindFirstChild("Head") then THead = TCharacter.Head end
    if TCharacter:FindFirstChildOfClass("Accessory") then Accessory = TCharacter:FindFirstChildOfClass("Accessory") end
    if Accessory and Accessory:FindFirstChild("Handle") then Handle = Accessory.Handle end
    if not (Character and Humanoid and RootPart) then return end
    local OldPos = nil
    if RootPart.Velocity.Magnitude < 50 then OldPos = RootPart.CFrame end
    if THumanoid and THumanoid.Sit then return end
    local FPos = function(BasePart, Pos, Ang)
        RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
        Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
        RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end
    local SFBasePart = function(BasePart)
        local TimeToWait = 2 local Time = tick() local Angle = 0
        repeat
            if RootPart and THumanoid then
                if BasePart.Velocity.Magnitude < 50 then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0)) task.wait()
                else
                    FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0)) task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0)) task.wait()
                end
            end
        until Time + TimeToWait < tick() or not FlingActive

local function UpdateAimLock()
    if not _S.AimLockEnabled then _S.AimLockLockedTarget = nil aimLockFovCircle.Visible = false return end
    if _S.ShowAimLockFOVCircle then aimLockFovCircle.Position = UserInputService:GetMouseLocation() aimLockFovCircle.Radius = _S.AimLockFOV aimLockFovCircle.Visible = true else aimLockFovCircle.Visible = false end
    if not _S.AimLockAlwaysRMB and not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then _S.AimLockLockedTarget = nil return end
    if _S.AimLockLockedTarget and _S.AimLockSkipDead and not IsPlayerAlive(_S.AimLockLockedTarget) then _S.AimLockLockedTarget = nil return end
    local targetStillValid = false
    if _S.AimLockLockedTarget and _S.AimLockLockedTarget.Parent and _S.AimLockLockedTarget.Character and IsAimLockTarget(_S.AimLockLockedTarget) then
        local t = _S.AimLockLockedTarget
        local isF = IsFriend(t)
        local head = t.Character and t.Character:FindFirstChild("Head")
        local los = head and (not _S.AimLockSkipWall or HasLineOfSight(head))
        if (not _S.AimLockSkipFriends or not isF) and los and head then targetStillValid = true end
    end
    if not targetStillValid then _S.AimLockLockedTarget = GetClosestTargetForAim() end
    local target = _S.AimLockLockedTarget
    if target and target.Character then
        local head = target.Character:FindFirstChild("Head")
        local hrp = target.Character:FindFirstChild("HumanoidRootPart")
        if head then
            local aimPos = head.Position
            if _S.AimLockPrediction and hrp then
                local uid = target.UserId local now = tick() local hist = _S.AimLockPredictionHistory
                if not hist[uid] then hist[uid] = { pos = hrp.Position, t = now } else
                    local dt = now - hist[uid].t
                    if dt > 0 and dt < 0.2 then local velocity = (hrp.Position - hist[uid].pos) / dt aimPos = head.Position + velocity * _S.AimLockPredictionStrength end
                    hist[uid] = { pos = hrp.Position, t = now }
                end
            end
            local targetCF = CFrame.new(Camera.CFrame.Position, aimPos)
            if _S.AimLockPreciseMode then Camera.CFrame = targetCF else Camera.CFrame = Camera.CFrame:Lerp(targetCF, 1 / math.max(_S.AimLockSmooth, 8)) end
            AutoFireAimLock(target)
        end
    end
end

-- ========== AIM TRACKER (Beam-based, всегда виден) ==========
local aimTrackActive = {}
local aimTrackData = {}

local function createAimBeam(plr)
    if not plr or not plr.Character then return end
    local char = plr.Character
    local head = char:FindFirstChild("Head")
    if not head then return end

    if aimTrackData[plr.UserId] and aimTrackData[plr.UserId].beam then
        pcall(function() aimTrackData[plr.UserId].beam:Destroy() end)
    end

    local beam = Instance.new("Beam")
    beam.Name = "AimTrackerBeam_" .. plr.UserId
    beam.Parent = char

    local attach0 = head:FindFirstChild("FaceCenterAttachment")
    if not attach0 then
        attach0 = Instance.new("Attachment", head)
        attach0.Name = "AimTrackerOrigin"
    end
    beam.Attachment0 = attach0

    local attach1 = Instance.new("Attachment", workspace.Terrain)
    beam.Attachment1 = attach1

    beam.Width0 = 0.15
    beam.Width1 = 0.15
    beam.FaceCamera = true
    beam.Transparency = NumberSequence.new(0.2)
    beam.LightEmission = 1
    beam.LightInfluence = 0
    beam.Color = ColorSequence.new(Color3.fromRGB(0, 255, 80))
    beam.Enabled = true

    aimTrackData[plr.UserId] = {
        beam = beam,
        attach1 = attach1,
    }

    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local mousePos = bodyEffects:FindFirstChild("MousePos")
        if mousePos then
            local function updateBeam()
                if not beam or not beam.Parent then return end
                local origin = attach0.WorldPosition
                local targetPos = mousePos.Value
                local dir = targetPos - origin

                local rp = RaycastParams.new()
                rp.FilterType = Enum.RaycastFilterType.Exclude
                rp.FilterDescendantsInstances = {char, Camera}
                rp.IgnoreWater = true

                local result = workspace:Raycast(origin, dir * 2, rp)
                local endPos = result and result.Position or targetPos

                attach1.WorldPosition = endPos

                local hitPlayer = false
                if result then
                    local hitModel = result.Instance:FindFirstAncestorWhichIsA("Model")
                    if hitModel then
                        for _, p in ipairs(Players:GetPlayers()) do
                            if p ~= plr and p.Character == hitModel then
                                hitPlayer = true
                                break
                            end
                        end
                    end
                end

                beam.Color = hitPlayer and ColorSequence.new(Color3.fromRGB(255, 40, 40))
                                       or ColorSequence.new(Color3.fromRGB(0, 255, 80))
            end

            updateBeam()
            mousePos.Changed:Connect(updateBeam)
        end
    end
end

local function clearAimTrack(plr)
    local d = aimTrackData[plr.UserId]
    if d then
        if d.beam then pcall(function() d.beam:Destroy() end) end
        if d.attach1 then pcall(function() d.attach1:Destroy() end) end
        aimTrackData[plr.UserId] = nil
    end
end

local function toggleAimTrack(plr)
    if aimTrackActive[plr.UserId] then
        aimTrackActive[plr.UserId] = nil
        clearAimTrack(plr)
    else
        aimTrackActive[plr.UserId] = true
        createAimBeam(plr)
    end
end

local function stopSpectate()
    if spectateSavedSubject then
        pcall(function()
            Camera.CameraSubject = spectateSavedSubject
            Camera.CameraType = spectateSavedType or Enum.CameraType.Custom
        end)
    else
        local myChar = LocalPlayer.Character
        if myChar then
            local myHum = myChar:FindFirstChildOfClass("Humanoid")
            if myHum then
                Camera.CameraSubject = myHum
                Camera.CameraType = Enum.CameraType.Custom
            end
        end
    end
    spectateTarget = nil
    spectateSavedSubject = nil
    spectateSavedType = nil
    local myChar = LocalPlayer.Character
    if myChar then
        for _, part in ipairs(myChar:GetDescendants()) do
            if part:IsA("BasePart") then
                part.LocalTransparencyModifier = 0
            elseif part:IsA("Decal") then
                part.Transparency = 0
            end
        end
    end
end

local function GetRageTarget()
    local closest, closestDist = nil, math.huge
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsLegitTarget(plr) and plr.Character then
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local d = (hrp.Position - myPos).Magnitude
                if d < closestDist then closestDist = d closest = plr end
            end
        end
    end
    return closest
end

local function DoStomp()
    local char = LocalPlayer.Character
    if not char then return end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
    end
    pcall(function()
        mouse1press()
        task.wait(0.005)
        mouse1release()
    end)
    pcall(function() mouse1click() end)
end

local function StopRageKill()
    _S.RageKillEnabled = false
    if _S.RageThread then pcall(function() task.cancel(_S.RageThread) end) _S.RageThread = nil end
    _S.RageTarget = nil
    _S.RageFinishPos = nil
    _S.RageLastTarget = nil
    local myChar = LocalPlayer.Character
    if myChar then local hum = myChar:FindFirstChildOfClass("Humanoid") if hum then hum.PlatformStand = false end end
end

local function StartRageKill()
    if _S.RageThread then pcall(function() task.cancel(_S.RageThread) end) end
    _S.RageAngle = 0
    _S.RageThread = task.spawn(function()
        while _S.RageKillEnabled do
            pcall(function()
                local myChar = LocalPlayer.Character
                if not myChar then return end
                local hrp = myChar:FindFirstChild("HumanoidRootPart")
                local hum = myChar:FindFirstChildOfClass("Humanoid")
                if not hrp or not hum or hum.Health <= 0 then task.wait(0.3) return end
                hum.PlatformStand = true

                local now = tick()

                if _S.RageFinishEnabled and _S.RageFinishPos and now < _S.RageFinishUntil then
                    local pos = _S.RageFinishPos
                    hrp.CFrame = CFrame.new(pos + Vector3.new(0, 1.5, 0))
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    if _S.RageAimCamEnabled then
                        local targetCF = CFrame.new(Camera.CFrame.Position, pos)
                        Camera.CFrame = Camera.CFrame:Lerp(targetCF, _S.RageAimCamSmooth)
                    end
                    if _S.RageFinishFastStomp then
                        for i = 1, 8 do DoStomp() end
                    else
                        DoStomp()
                    end
                    task.wait(0.01)
                    return
                end

                local target = GetRageTarget()

                if target and target.Character then
                    _S.RageTarget = target
                    _S.RageLastTarget = target
                    local targetHrp = target.Character:FindFirstChild("HumanoidRootPart")
                    local targetHum = target.Character:FindFirstChildOfClass("Humanoid")
                    if targetHrp and targetHum and targetHum.Health > 0 then
                        local targetPos = targetHrp.Position
                        _S.RageAngle = _S.RageAngle + 0.7
                        local newPos = targetPos + Vector3.new(
                            math.cos(_S.RageAngle) * _S.RageRadius,
                            _S.RageHeight,
                            math.sin(_S.RageAngle) * _S.RageRadius
                        )
                        hrp.CFrame = CFrame.new(newPos, targetPos)
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero

                        if _S.RageAimCamEnabled then
                            local head = target.Character:FindFirstChild("Head")
                            local aimPos = head and head.Position or (targetPos + Vector3.new(0, 1.5, 0))
                            local targetCF = CFrame.new(Camera.CFrame.Position, aimPos)
                            Camera.CFrame = Camera.CFrame:Lerp(targetCF, _S.RageAimCamSmooth)
                        end

                        if _S.RageAutoShoot then
                            DoStomp()
                        end
                    end
                else
                    if _S.RageFinishEnabled and _S.RageLastTarget and _S.RageLastTarget.Character then
                        local lastHum = _S.RageLastTarget.Character:FindFirstChildOfClass("Humanoid")
                        local lastHrp = _S.RageLastTarget.Character:FindFirstChild("HumanoidRootPart")
                        if lastHum and lastHrp and lastHum.Health <= 0 then
                            _S.RageFinishPos = lastHrp.Position
                            _S.RageFinishUntil = now + _S.RageFinishTime
                            _S.RageLastTarget = nil
                            return
                        end
                    end
                end
            end)
            task.wait(_S.RageSpeed)
        end
    end)
end

local function updateRoleESPForPlayer(player)
    local o = roleEspObjects[player.UserId] if not o then return end
    if not shouldShowRoleESP(player) then
        if o.nameLabel then o.nameLabel.Visible = false end if o.distLabel then o.distLabel.Visible = false end
        if o.box then o.box.Visible = false end if o.boxOutline then o.boxOutline.Visible = false end
        if o.tracer then o.tracer.Visible = false end if o.hbar then o.hbar.Visible = false end if o.hbarOutline then o.hbarOutline.Visible = false end
        return
    end
    local char = player.Character if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid") local rp = char:FindFirstChild("UpperTorso") or char:FindFirstChild("HumanoidRootPart") local hd = char:FindFirstChild("Head")
    if not rp then return end
    local rc = _S.VisualColors.RoleESP[GetPlayerRole(player)] or Color3.fromRGB(160,160,180)
    local rSP, rOn = Camera:WorldToViewportPoint(rp.Position) local nSP, nOn = Camera:WorldToViewportPoint(rp.Position + Vector3.new(0, 2.5, 0)) local dist = (Camera.CFrame.Position - rp.Position).Magnitude
    if o.nameLabel then
        if roleEspSettings.ShowNames and nOn then
            local role = GetPlayerRole(player) local icons = {Target="🎯", Priority="⚡", Friend="💚", Neutral="⚪"}
            o.nameLabel.Text = (icons[role] or "").." "..player.Name o.nameLabel.TextColor3 = rc o.nameLabel.Position = UDim2.new(0, nSP.X - 90, 0, nSP.Y - 28) o.nameLabel.Visible = true
        else o.nameLabel.Visible = false end
    end
    if o.distLabel then
        if roleEspSettings.ShowDistance and nOn then o.distLabel.Text = string.format("%.0f m", dist) o.distLabel.TextColor3 = rc o.distLabel.Position = UDim2.new(0, nSP.X - 90, 0, nSP.Y - 12) o.distLabel.Visible = true else o.distLabel.Visible = false end
    end
    if hd and rOn then
        local hp = Camera:WorldToViewportPoint(hd.Position + Vector3.new(0, 0.5, 0)) local fp = Camera:WorldToViewportPoint(rp.Position - Vector3.new(0, 3, 0))
        local bh = math.max(math.abs(hp.Y - fp.Y), 10) local bw = bh / 2 local bx = hp.X - bw / 2 local by = hp.Y - bh * 0.1
        if roleEspSettings.ShowBox then o.box.Size = Vector2.new(bw, bh) o.box.Position = Vector2.new(bx, by) o.box.Color = rc o.box.Transparency = 1 o.box.Visible = true o.boxOutline.Size = o.box.Size o.boxOutline.Position = o.box.Position o.boxOutline.Transparency = 1 o.boxOutline.Visible = true else o.box.Visible = false o.boxOutline.Visible = false end
        if roleEspSettings.ShowHealthBar and hum then
            local mh = math.max(hum.MaxHealth or 1, 1) local hr = math.clamp(hum.Health/mh,0,1) local hbh = bh * hr
            local hc = Color3.fromRGB(math.floor((1-hr)*255), math.floor(hr*220), 50)
            o.hbar.Size = Vector2.new(3, hbh) o.hbar.Position = Vector2.new(bx-7, by+(bh-hbh)) o.hbar.Color = hc o.hbar.Transparency = 1 o.hbar.Visible = true
            o.hbarOutline.Size = Vector2.new(3, bh) o.hbarOutline.Position = Vector2.new(bx-7, by) o.hbarOutline.Transparency = 1 o.hbarOutline.Visible = true
        else o.hbar.Visible = false o.hbarOutline.Visible = false end
    else
        if o.box then o.box.Visible = false end if o.boxOutline then o.boxOutline.Visible = false end
        if o.hbar then o.hbar.Visible = false end if o.hbarOutline then o.hbarOutline.Visible = false end
    end
    if roleEspSettings.ShowTracers and rOn then o.tracer.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y) o.tracer.To = Vector2.new(rSP.X, rSP.Y) o.tracer.Color = rc o.tracer.Transparency = 1 o.tracer.Visible = true else if o.tracer then o.tracer.Visible = false end end
end

local flyEnabled = false local flySpeed = 50 local flyConnection = nil
local flyBodyVelocity, flyBodyGyro = nil, nil
local function stopFly() flyEnabled = false if flyConnection then flyConnection:Disconnect() flyConnection = nil end if flyBodyVelocity and flyBodyVelocity.Parent then flyBodyVelocity:Destroy() flyBodyVelocity = nil end if flyBodyGyro and flyBodyGyro.Parent then flyBodyGyro:Destroy() flyBodyGyro = nil end local char = LocalPlayer.Character if char then local hum = char:FindFirstChildOfClass("Humanoid") if hum then hum.PlatformStand = false end end end
local function startFly()
    flyEnabled = true local char = LocalPlayer.Character if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart") local hum = char:FindFirstChildOfClass("Humanoid") if not hrp or not hum then return end
    hum.PlatformStand = true
    flyBodyVelocity = Instance.new("BodyVelocity") flyBodyVelocity.Velocity = Vector3.zero flyBodyVelocity.MaxForce = Vector3.new(1e5,1e5,1e5) flyBodyVelocity.Parent = hrp
    flyBodyGyro = Instance.new("BodyGyro") flyBodyGyro.MaxTorque = Vector3.new(1e5,1e5,1e5) flyBodyGyro.P = 1e4 flyBodyGyro.CFrame = hrp.CFrame flyBodyGyro.Parent = hrp
    flyConnection = RunService.RenderStepped:Connect(function()
        if not flyEnabled then return end
        local camCF = workspace.CurrentCamera.CFrame local md = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then md = md + camCF.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then md = md - camCF.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then md = md - camCF.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then md = md + camCF.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then md = md + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then md = md - Vector3.new(0,1,0) end
        if flyBodyVelocity and flyBodyVelocity.Parent then flyBodyVelocity.Velocity = md.Magnitude > 0 and md.Unit * flySpeed or Vector3.zero end
        if flyBodyGyro and flyBodyGyro.Parent then flyBodyGyro.CFrame = camCF end
    end)
end

createToggle(MovementContent,"Fly",false,function(s) if s then startFly() else stopFly() end end)

    end

UserInputService.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == Enum.KeyCode.RightShift or i.KeyCode == Enum.KeyCode.End then toggleMenu() return end
    local function acc(v) if v.UserInputType == Enum.UserInputType.Keyboard then if v.KeyCode ~= Enum.KeyCode.Unknown then return v.KeyCode end elseif MOUSE_BUTTON_TYPES[v.UserInputType] then return v.UserInputType end return nil end
    local nb = acc(i)
    if _S.WaitingForRageBind and nb then
        _S.RageBindKey = nb
        rageBindBtn.Text = getBindName(nb)
        rageBindBtn.BackgroundColor3 = _S.Theme.BindBtn
        _S.WaitingForRageBind = false
        return
    end
    if _S.RageBindKey and isBindMatch(_S.RageBindKey, i) then
        _S.RageKillEnabled = not _S.RageKillEnabled
        if _S.RageKillEnabled then StartRageKill() else StopRageKill() end
        if rageSetVal then rageSetVal(_S.RageKillEnabled, true) end
    end
    if _S.WaitingForAimLockBind and nb then _S.AimLockBindKey = nb _S.WaitingForAimLockBind = false return end
    if _S.WaitingForMouseLockBind and nb then _S.MouseLockBindKey = nb mb.Text = "Бинд: " .. getBindName(nb) mb.BackgroundColor3 = Color3.fromRGB(0,120,180) _S.WaitingForMouseLockBind = false return end
    if _S.WaitingForTriggerBind and nb then _S.TriggerBindKey = nb if _S.TriggerKeyBindBtnUpdate then _S.TriggerKeyBindBtnUpdate(nb) end _S.WaitingForTriggerBind = false return end
    if _S.AimLockBindKey and isBindMatch(_S.AimLockBindKey, i) then
        if _S.AimLockBindMode == "Toggle" then
            _S.AimLockEnabled = not _S.AimLockEnabled
            if _S.AimLockEnabled then if not _S.AimLockConnection then _S.AimLockConnection = RunService.RenderStepped:Connect(UpdateAimLock) end else _S.AimLockLockedTarget = nil aimLockFovCircle.Visible = false if _S.AimLockConnection then _S.AimLockConnection:Disconnect() _S.AimLockConnection = nil end end
        elseif _S.AimLockBindMode == "Hold" then
            _S.AimLockEnabled = true
            if not _S.AimLockConnection then _S.AimLockConnection = RunService.RenderStepped:Connect(UpdateAimLock) end
        end
    end
    if _S.MouseLockBindKey and isBindMatch(_S.MouseLockBindKey, i) then
        if _S.MouseLockBindMode == "Toggle" then
            _S.MouseLockEnabled = not _S.MouseLockEnabled
            if _S.MouseLockEnabled then if not _S.MouseLockConnection then _S.MouseLockConnection = RunService.RenderStepped:Connect(UpdateMouseLock) end else _S.MouseLockLockedTarget = nil mouseLockFovCircle.Visible = false if _S.MouseLockConnection then _S.MouseLockConnection:Disconnect() _S.MouseLockConnection = nil end end
        elseif _S.MouseLockBindMode == "Hold" then _S.MouseLockEnabled = true if not _S.MouseLockConnection then _S.MouseLockConnection = RunService.RenderStepped:Connect(UpdateMouseLock) end end
    end
    if _S.TriggerBindKey and isBindMatch(_S.TriggerBindKey, i) then
        if _S.TriggerKeyBindMode == "Toggle" then
            _S.TriggerbotEnabled = not _S.TriggerbotEnabled
            UpdateTriggerbot()
        elseif _S.TriggerKeyBindMode == "Hold" then
            _S.TriggerbotEnabled = true
            UpdateTriggerbot()
        end
    end
end)
